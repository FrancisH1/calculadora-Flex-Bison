// Constantes e funcoes cientificas

let e = 2.71828

// Teste de Notacao Cientifica
let n1 = 1.5e+2

print(n1 + n2)

// Calculos com funcoes internas
let a = sqrt(144)
let b = log(e)
let c = exp(2)

// Expressao de "estresse" para o parser
let formula = (a * b + c) / (sqrt(PI) * 2) - 0.001
print(formula)

// Atribuicao e comparacao em uma linha
let check = (a = 10) == 10
